cmake_minimum_required(VERSION 3.20)
project(paketbote LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Qml Quick QuickControls2)
set(QML_ZIEL "${CMAKE_INSTALL_PREFIX}/share/paketbote")
add_executable(paketbote main.cpp)
target_compile_definitions(paketbote PRIVATE PAKETBOTE_QML="${QML_ZIEL}/app/main.qml")
target_link_libraries(paketbote PRIVATE Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS paketbote DESTINATION bin)
install(DIRECTORY ../app ../gemeinsam DESTINATION share/paketbote)
install(FILES de.teyro.Paketbote.desktop DESTINATION share/applications)
install(FILES ../paketbote.svg DESTINATION share/icons/hicolor/scalable/apps RENAME de.teyro.Paketbote.svg)
install(FILES de.teyro.Paketbote.metainfo.xml DESTINATION share/metainfo)
install(FILES ../paketbote.notifyrc DESTINATION share/knotifications6 RENAME paketbote.notifyrc)
